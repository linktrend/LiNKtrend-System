### Static audit contract validation

Command:

```bash
python3 - <<'PY'
from pathlib import Path
path = Path('supabase/migrations/001_linkbrain_audit_envelope.sql')
sql = path.read_text()
required = [
    'run.started', 'run.completed', 'run.failed',
    'stage.started', 'stage.completed', 'stage.failed',
    'lease.requested', 'lease.granted', 'lease.executed', 'lease.denied',
    'workflow.invoked', 'workflow.completed', 'workflow.failed',
    'create table if not exists linkbrain.audit_events',
    'create table if not exists linkbrain.memory_objects',
    'create policy audit_events_tenant_select',
    'create policy memory_objects_tenant_select',
    'create or replace function linkbrain.write_audit_event',
    'create or replace function linkbrain.write_memory_object',
]
missing = [item for item in required if item not in sql]
if missing:
    raise SystemExit(f'Missing expected audit contract tokens: {missing}')
print(f'{path}: static audit contract tokens present ({len(required)} checks)')
PY
```
